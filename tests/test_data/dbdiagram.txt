Table tests_Customer {
  id "integer" [primary key]
  first_name "text"
  last_name "text"
}
Table tests_Product {
  id "integer" [primary key]
  sku "text"
  product_name "text"
  product_code "text"
  quantity "integer"
  price "decimal"
}
Table tests_Order {
  id "integer" [primary key]
  customer_id "integer"
  product_id "integer"
  quantity "integer"
  order_total "decimal"
}
Table tests_Region {
  id "integer" [primary key]
  name "text"
  label "text"
}
Ref: tests_Product.id <> tests_Region.id
Ref: tests_Order.customer_id > tests_Customer.id
Ref: tests_Order.product_id > tests_Product.id
